# Frame Enrollment and Two-Port Admin — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Frames pull their per-device keys at setup through a single-use enrollment code, ship zero secrets in the artifact, and split traffic between a localhost display port and a LAN management port.

**Architecture:** The console stores hashed enrollment codes and mints the device token only inside the atomic redeem transaction; `POST /api/enroll` returns the frame's key bundle. On the frame, `kiosk-ctl enroll` writes the bundle to `/etc/dscos/frame.env` (0640, before the build so `NEXT_PUBLIC_*` inlines). `instrumentation.ts` opens a TCP forwarder on `KIOSK_MGMT_PORT` (default 8808) into the app bound to `127.0.0.1:3000`, and the proxy layer gates routes by the `Host` port plus a frame-local HMAC cookie for mutations.

**Tech Stack:** Next.js 16 (App Router, `proxy.ts` middleware), TypeScript strict, Drizzle (Postgres + better-sqlite3 dual dialect), Vitest, nanoid, node:crypto, bash installer (`install-kiosk.sh`).

**Spec:** `docs/superpowers/specs/2026-10-25-frame-enrollment-design.md`

**Repo checks:** `npm run test` (vitest run), `npm run lint`, `npm run build`.

Before Tasks 7 and 8 (proxy/middleware and instrumentation), read the relevant guides in `node_modules/next/dist/docs/` (see `01-app/` — search for "proxy"/"middleware" and "instrumentation") and follow what the bundled docs say, not memory.

```ts
// Fixed-window in-memory limiter for /api/enroll and /api/admin/login.
// Per-process best-effort: a horizontally scaled console counts per instance.
// Enrollment code entropy covers what an imperfect limiter misses (spec §3.2).
type Window = { count: number; resetAt: number }
const windows = new Map<string, Window>()

export interface LimitResult { allowed: boolean; retryAfterMs: number }

export function checkLimit(key: string, max: number, windowMs: number, now: () => number = Date.now): LimitResult {
  const t = now()
  const w = windows.get(key)
  if (!w || t >= w.resetAt) {
    windows.set(key, { count: 1, resetAt: t + windowMs })
    return { allowed: true, retryAfterMs: 0 }
  }
  w.count += 1
  if (w.count > max) return { allowed: false, retryAfterMs: w.resetAt - t }
  return { allowed: true, retryAfterMs: 0 }
}

// Drop stale windows so the map cannot grow without bound. Callers on the
// request path invoke this cheaply; keys are bounded either way.
export function sweepLimits(now: () => number = Date.now): void {
  const t = now()
  for (const [k, w] of windows) if (t >= w.resetAt) windows.delete(k)
}

export function __resetLimits(): void { windows.clear() }
```

```ts
import crypto from 'node:crypto'
import { and, eq, gt, isNull } from 'drizzle-orm'
import { nanoid } from 'nanoid'
import { getDb } from '@/lib/db'
import { getTables } from '@/lib/db/tables'
import { hashToken } from '@/lib/sync/auth'

export interface IssuedCode { instanceId: string; code: string; expiresAt: Date }
export interface EnrollBundle {
  instanceId: string
  deviceToken: string
  frameAdminSecret: string
  clerkPublishableKey: string
}

const CODE_TTL_MS = 15 * 60 * 1000

/** Create the instance row (token hash stays NULL until redeem) + seed frame doc. */
export async function issueEnrollmentCode(instanceId?: string, ttlMs: number = CODE_TTL_MS): Promise<IssuedCode> {
  const db = getDb()
  const t = getTables()
  const now = new Date()
  let id = instanceId
  if (!id) {
    id = nanoid(12)
    await db.insert(t.instances).values({ id, deviceTokenHash: null, createdAt: now })
    const existing = await db.select({ id: t.frameDocuments.id }).from(t.frameDocuments)
      .where(eq(t.frameDocuments.identifier, id))
    if (!existing.length) {
      const { seed } = await import('@/lib/store')
      await db.insert(t.frameDocuments).values({
        id, identifier: id, name: `Frame ${id}`, data: seed(),
        createdAt: now, updatedAt: now, revision: 1, resolutionWidth: 1920, resolutionHeight: 1080,
      })
    }
  }
  const code = nanoid(20)
  const expiresAt = new Date(now.getTime() + ttlMs)
  await db.insert(t.enrollmentCodes).values({ codeHash: hashToken(code), instanceId: id, createdAt: now, expiresAt })
  return { instanceId: id, code, expiresAt }
}

/**
 * Atomic single-use redeem. The plaintext device token is minted here and
 * returned; it is never stored. Null covers unknown/expired/consumed codes and
 * deleted instances alike (spec §5: one generic failure).
 */
export async function redeemEnrollmentCode(code: string): Promise<EnrollBundle | null> {
  const db = getDb()
  const t = getTables()
  const codeHash = hashToken(code)
  const now = new Date()

  const claimed = await db.update(t.enrollmentCodes)
    .set({ consumedAt: now })
    .where(and(eq(t.enrollmentCodes.codeHash, codeHash), isNull(t.enrollmentCodes.consumedAt), gt(t.enrollmentCodes.expiresAt, now)))
    .returning({ instanceId: t.enrollmentCodes.instanceId })

  const row = claimed[0] as { instanceId: string } | undefined
  if (!row) return null

  const instanceId = row.instanceId
  const deviceToken = nanoid(32)
  const updated = await db.update(t.instances)
    .set({ deviceTokenHash: hashToken(deviceToken) })
    .where(eq(t.instances.id, instanceId))
  if (!(updated as { rowsAffected?: number }).rowsAffected) {
    // Instance revoked between issue and redeem: burn the code, fail generically.
    return null
  }

  return {
    instanceId,
    deviceToken,
    frameAdminSecret: crypto.randomBytes(32).toString('base64url'),
    clerkPublishableKey: process.env.NEXT_PUBLIC_CLERK_PUBLISHABLE_KEY ?? '',
  }
}
```

`returning()` and `rowsAffected` behave the same on both drivers through the shared drizzle API; if the PG update result shape differs, normalize inside this module only (the two casts in `tables.ts` are the sanctioned dialect seams).

```ts
/**
 * Register a device. Enrollment moved to lib/console/enroll: the admin API
 * returns only a one-time code now, and the plaintext token exists solely in
 * the /api/enroll response (spec §3).
 */
export async function registerInstance(name?: string): Promise<{ instanceId: string; enrollmentCode: string; expiresAt: Date }> {
  const { issueEnrollmentCode } = await import('@/lib/console/enroll')
  const issued = await issueEnrollmentCode()
  if (name) {

    await db.update(tables.frameDocuments)

      .where(eq(tables.frameDocuments.identifier, issued.instanceId))
  }
  return { instanceId: issued.instanceId, enrollmentCode: issued.code, expiresAt: issued.expiresAt }
}
```

```ts
// Frame port-gate decisions, spec §4.2. Pure: takes plain strings so vitest
// can test it without Next request plumbing. proxy.ts adapts req → input.
// Active only when KIOSK_MGMT_PORT is set (frames). The app binds
// 127.0.0.1:PORT (display) and the forwarder exposes 0.0.0.0:MGMT; a request
// whose Host carries any other port cannot have arrived on a listener we
// intend, so it is rejected.

export type GateInput = { host: string; method: string; path: string }
export type GateDecision =
  | { kind: 'allow' }
  | { kind: 'not_found' }
  | { kind: 'admin_required' }

const DISPLAY_GET_ONLY = ['/api/active', '/api/store', '/api/assets', '/api/health', '/api/media', '/api/fonts', '/api/display']
const FRAME_NEVER = ['/console', '/api/instances', '/api/enroll'] // console-only surfaces; never served by a frame
const ADMIN_EXEMPT_MUTATIONS = ['/api/sync', '/api/admin/login']

export function hostPort(host: string): string | null {
  const m = host.match(/:(\d+)$/)
  return m ? m[1] : null
}

export function decideRequest(input: GateInput): GateDecision {
  const mgmt = process.env.KIOSK_MGMT_PORT
  if (!mgmt) return { kind: 'allow' } // console deployments: gate inert (spec §4.2)

  const { path, method, host } = input
  const write = method !== 'GET' && method !== 'HEAD'

  if (FRAME_NEVER.some((p) => path === p || path.startsWith(`${p}/`))) return { kind: 'not_found' }

  const port = hostPort(host)
  if (port === mgmt) {
    if (!write) return { kind: 'allow' }
    if (ADMIN_EXEMPT_MUTATIONS.some((p) => path === p || path.startsWith(`${p}/`))) return { kind: 'allow' }
    return { kind: 'admin_required' }
  }

  const appPort = process.env.PORT || '3000'
  if (port !== null && port !== appPort) return { kind: 'not_found' }
  // Display listener: frame pages + their GET feeds. Everything else 404s.
  if (path.startsWith('/frame')) return { kind: 'allow' }
  if (path === '/sw.js') return { kind: 'allow' }
  if (!write && DISPLAY_GET_ONLY.some((p) => path === p || path.startsWith(`${p}/`))) return { kind: 'allow' }
  return { kind: 'not_found' }
}
```

- `/` (the editor) is intentionally `not_found` on the display port; the editor is reached via `:8808`. The kiosk browser only opens `/frame/<id>`.
- `/api/frames` mutations are excluded on display by the write rule; `/api/frames` GET also isn't on the display list because the display page reads `/api/active` + `/api/store`, not the CRUD list. Verify against `app/frame/[id]` page's fetches before finalizing the list (open the page file, collect every `/api/` request it or its components make, and add the read-only ones to `DISPLAY_GET_ONLY`).
- `/api/enroll` lives in `FRAME_NEVER` because only the console redeems codes; a frame hosting the endpoint itself would be a loop.

```ts
import crypto from 'node:crypto'

// Frame-local admin auth (spec §4.3): scrypt password hash in frame.env, an
// HMAC-signed stateless session cookie for management-port mutations. No
// server-side session store: local-first means auth survives console and
// internet outages.

export const ADMIN_COOKIE = 'dscos_admin'
const SESSION_TTL_MS = 7 * 24 * 3600_000
const SCRYPT = { N: 32768, r: 8, p: 1 }

export function hashAdminPassword(secret: string): string {
  const salt = crypto.randomBytes(16).toString('hex')
  const hash = crypto.scryptSync(secret, salt, 32, SCRYPT).toString('hex')
  return `${salt}$${hash}`
}

export function verifyAdminPassword(secret: string, stored: string): boolean {
  const [salt, hex] = stored.split('$')
  if (!salt || !hex) return false
  const candidate = crypto.scryptSync(secret, salt, 32, SCRYPT)
  const expected = Buffer.from(hex, 'hex')
  return expected.length === candidate.length && crypto.timingSafeEqual(expected, candidate)
}

export function signAdminSession(secret: string, nowMs: number = Date.now()): string {
  const exp = nowMs + SESSION_TTL_MS
  const payload = `admin.${exp}`
  const sig = crypto.createHmac('sha256', secret).update(payload).digest('base64url')
  return `${payload}.${sig}`
}

export function verifyAdminSession(token: string | undefined, secret: string | undefined, nowMs: number = Date.now()): boolean {
  if (!token || !secret) return false
  const m = token.match(/^admin\.(\d+)\.([A-Za-z0-9_-]+)$/)
  if (!m) return false
  const exp = Number(m[1])
  if (!Number.isFinite(exp) || exp < nowMs) return false
  const expected = crypto.createHmac('sha256', secret).update(`admin.${m[1]}`).digest('base64url')
  const a = Buffer.from(expected), b = Buffer.from(m[2])
  return a.length === b.length && crypto.timingSafeEqual(a, b)
}

// True when a management-port mutation may proceed. Cookie must match the
// minted secret; FRAME_ADMIN_HASH unset means read-only (spec §4.3).
export function requestHasAdmin(headers: Headers): boolean {
  const cookie = headers.get('cookie') ?? ''
  const token = cookie.split('; ').find((c) => c.startsWith(`${ADMIN_COOKIE}=`))?.slice(ADMIN_COOKIE.length + 1)
  return verifyAdminSession(token, process.env.FRAME_ADMIN_SECRET)
}

export function adminAuthEnabled(): boolean {
  return Boolean(process.env.FRAME_ADMIN_HASH)
}
```

```ts
import net from 'node:net'

// Spec §4.2: the only thing listening on the LAN. Raw TCP (no HTTP parsing)
// keeps the client's Host header intact so proxy.ts can gate by port.
// Failures bind-critical: a busy port logs loudly, it must not kill the app.

export interface ForwarderOptions { mgmtPort: number; appPort: number; bind?: string }

export function startMgmtForwarder(opts: ForwarderOptions): Promise<net.Server> {
  return new Promise((resolve, reject) => {
    const server = net.createServer((incoming) => {
      const upstream = net.connect(opts.appPort, '127.0.0.1')
      incoming.pipe(upstream)
      upstream.pipe(incoming)
      const drop = () => { incoming.destroy(); upstream.destroy() }
      incoming.on('error', drop)
      upstream.on('error', drop)
    })
    server.once('error', reject)
    server.listen(opts.mgmtPort, opts.bind ?? '0.0.0.0', () => resolve(server))
  })
}

// Called from instrumentation.ts on the node runtime. Console deployments
// leave KIOSK_MGMT_PORT unset and get nothing.
export async function maybeStartMgmtForwarder(env: NodeJS.ProcessEnv): Promise<net.Server | null> {
  if (!env.KIOSK_MGMT_PORT) return null
  try {
    return await startMgmtForwarder({ mgmtPort: Number(env.KIOSK_MGMT_PORT), appPort: Number(env.PORT || 3000) })
  } catch (e: unknown) {
    console.error('management port forwarder failed to start:', e)
    return null
  }
}
```

```ts
import fs from 'node:fs'
import path from 'node:path'

// /etc/dscos/frame.env: the frame's pulled config. Root owns it, the kiosk
// service user reads it (0640). Values here must never be quoted-escaped
// payloads beyond simple KEY=VALUE tokens; enroll bundles and scrypt hashes
// qualify.

export function frameEnvLines(vars: Record<string, string>): string {
  return Object.entries(vars).map(([k, v]) => `${k}=${v}`).join('\n') + '\n'
}

export function parseFrameEnv(text: string): Record<string, string> {
  const out: Record<string, string> = {}
  for (const line of text.split(/\r?\n/)) {
    const m = line.match(/^([A-Z0-9_]+)=(.*)$/)
    if (m) out[m[1]] = m[2]
  }
  return out
}

export function writeFrameEnv(file: string, vars: Record<string, string>): void {
  fs.mkdirSync(path.dirname(file), { recursive: true })
  const tmp = `${file}.tmp`
  fs.writeFileSync(tmp, frameEnvLines(vars), { mode: 0o640 })
  fs.renameSync(tmp, file)
}

export function setFrameEnvValue(file: string, key: string, value: string): void {
  const existing = fs.existsSync(file) ? parseFrameEnv(fs.readFileSync(file, 'utf8')) : {}
  existing[key] = value
  writeFrameEnv(file, existing)
}

export function frameEnvPath(): string {
  return process.env.FRAME_ENV_PATH || '/etc/dscos/frame.env'
}
```

```ts
import fs from 'node:fs'
import { execSync } from 'node:child_process'
import { writeFrameEnv, frameEnvPath } from './frame-env'

export interface EnrollResponse {
  instanceId: string
  deviceToken: string
  syncEndpoint: string
  frameAdminSecret: string
  config?: { clerkPublishableKey?: string }
}

export interface EnrollDeps {
  fetch: typeof fetch
  envPath?: string
}

export async function enroll(consoleUrl: string, code: string, deps: EnrollDeps): Promise<{ ok: boolean; status?: number; message?: string; written?: string }> {
  if (!/^https?:\/\//.test(consoleUrl)) return { ok: false, message: 'console URL must be http(s)' }
  let res: Response
  try {
    res = await deps.fetch(`${consoleUrl.replace(/\/$/, '')}/api/enroll`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ code }),
    })
  } catch {
    return { ok: false, message: 'console unreachable; check the URL and network' }
  }
  if (!res.ok) {
    const hint = res.status === 401 ? 'code invalid, expired, or already used; issue a fresh one from the console (rotate for an existing instance)'
      : res.status === 429 ? 'rate limited; wait a minute and retry'
      : `enroll failed (HTTP ${res.status})`
    return { ok: false, status: res.status, message: hint }
  }
  const bundle = (await res.json()) as EnrollResponse
  const vars: Record<string, string> = {
    FRAME_MODE: 'instance',
    SYNC_ENABLED: 'true',
    SYNC_ENDPOINT: bundle.syncEndpoint,
    INSTANCE_ID: bundle.instanceId,
    SYNC_TOKEN: bundle.deviceToken,
    FRAME_ADMIN_SECRET: bundle.frameAdminSecret,
  }
  if (bundle.config?.clerkPublishableKey) vars.NEXT_PUBLIC_CLERK_PUBLISHABLE_KEY = bundle.config.clerkPublishableKey
  const file = deps.envPath ?? frameEnvPath()
  writeFrameEnv(file, vars)
  chownToKioskGroup(file)
  return { ok: true, written: file }
}

// Best-effort root:kiosk ownership so the service user can read the file
// while the world cannot. Non-root dev machines (tests) keep it as-is.
function chownToKioskGroup(file: string): void {
  if (typeof process.getuid === 'function' && process.getuid() !== 0) return
  try {
    const gid = Number(execSync('id -g kiosk 2>/dev/null || id -g', { encoding: 'utf8' }).trim()) || 0
    fs.chownSync(file, 0, gid)
    fs.chmodSync(file, 0o640)
  } catch { /* keep the 0640 from writeFrameEnv */ }
}
```
